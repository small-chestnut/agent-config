  ```python
  """入出金1件の消込の状態を返す。"""


  def clearing_status(payment_amount, allocations, imported_ids):
      """payment_amount: 入出金の額。
      allocations: 先方のデータにある、その入出金の割当の行(id と amount)。

      if not allocations:

      return "一部消込"
  ```
